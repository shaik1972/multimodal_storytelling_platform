Replace src/orchestration/orchestrator.py

from langchain_core.prompts import PromptTemplate
from langchain_core.output_parsers import StrOutputParser
from langchain_ibm import WatsonxLLM
from src.audio.audio_pipeline import generate_voice
from src.image.image_pipeline import generate_image
from src.video.video_pipeline import generate_video
from src import config

def _llm():
    return WatsonxLLM(
        model_id=config.GRANITE_MODEL_ID,
        url=config.IBM_URL,
        apikey=config.IBM_API_KEY,
        project_id=config.IBM_PROJECT_ID,
        params={"max_new_tokens": 500, "temperature": 0.8},
    )

story_prompt = PromptTemplate.from_template(
    "Write a vivid short story from this idea:\n\n{idea}\n\nStory:"
)
image_prompt_chain = PromptTemplate.from_template(
    "Summarise this story as a single visual scene description "
    "suitable for an image generator (max 30 words):\n\n{story}"
)

def run_full_pipeline(story_idea: str) -> dict:
    llm = _llm()
    parser = StrOutputParser()

    # 1. Generate story text
    story_text = (story_prompt | llm | parser).invoke({"idea": story_idea})

    # 2. Generate image prompt from story
    image_desc = (image_prompt_chain | llm | parser).invoke({"story": story_text})

    # 3. Generate outputs
    audio_bytes = generate_voice(story_text)
    image_bytes = generate_image(image_desc)
    video_bytes = generate_video(image_desc)

    return {
        "story_text":  story_text,
        "image_desc":  image_desc,
        "audio_bytes": audio_bytes,
        "image_bytes": image_bytes,
        "video_bytes": video_bytes,
    }

import io, tempfile, os
from moviepy.editor import ImageClip, AudioFileClip, CompositeVideoClip, TextClip

def generate_video(prompt: str, image_bytes: bytes = b"", audio_bytes: bytes = b"", duration: int = 10) -> bytes:
    """Compose a video from image + audio bytes."""
    import tempfile

    with tempfile.NamedTemporaryFile(suffix=".png", delete=False) as img_f:
        img_f.write(image_bytes or b"")
        img_path = img_f.name

    with tempfile.NamedTemporaryFile(suffix=".mp3", delete=False) as aud_f:
        aud_f.write(audio_bytes or b"")
        aud_path = aud_f.name

    clip = ImageClip(img_path).set_duration(duration)
    if audio_bytes:
        audio = AudioFileClip(aud_path)
        clip = clip.set_audio(audio)

    out_path = tempfile.mktemp(suffix=".mp4")
    clip.write_videofile(out_path, fps=24, codec="libx264", logger=None)

    with open(out_path, "rb") as f:
        video_bytes = f.read()

    for p in [img_path, aud_path, out_path]:
        try: os.remove(p)
        except: pass

    return video_bytes

from fastapi import FastAPI
from fastapi.middleware.cors import CORSMiddleware
from pydantic import BaseModel
from src.orchestration.orchestrator import run_full_pipeline
from src.utils.file_utils import save_text, save_audio, save_image, save_video

app = FastAPI(title="Multimodal Storytelling API")
app.add_middleware(CORSMiddleware, allow_origins=["*"], allow_methods=["*"], allow_headers=["*"])

class StoryRequest(BaseModel):
    idea: str

@app.post("/generate")
def generate_story(req: StoryRequest):
    result = run_full_pipeline(req.idea)
    save_text(result["story_text"])
    save_audio(result["audio_bytes"])
    save_image(result["image_bytes"])
    save_video(result["video_bytes"])
    return {
        "story_text": result["story_text"],
        "image_desc": result["image_desc"],
        "text_url":  "/outputs/story.txt",
        "audio_url": "/outputs/story.mp3",
        "image_url": "/outputs/story.png",
        "video_url": "/outputs/story.mp4",
    }

@app.get("/health")
def health(): return {"status": "ok"}

"use client";
import { useState } from "react";

export default function Home() {
  const [idea, setIdea] = useState("");
  const [loading, setLoading] = useState(false);
  const [result, setResult] = useState<any>(null);

  async function generate() {
    setLoading(true);
    const res = await fetch("http://localhost:8000/generate", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ idea }),
    });
    setResult(await res.json());
    setLoading(false);
  }

  return (
    <main className="max-w-2xl mx-auto p-8">
      <h1>Multimodal Storyteller</h1>
      <textarea value={idea} onChange={e => setIdea(e.target.value)}
        placeholder="Enter your story idea..." rows={4} />
      <button onClick={generate} disabled={loading}>
        {loading ? "Generating..." : "Generate Story"}
      </button>
      {result && (
        <div>
          <h2>Story</h2><p>{result.story_text}</p>
          <img src={result.image_url} alt="generated scene" />
          <audio controls src={result.audio_url} />
          <video controls src={result.video_url} />
        </div>
      )}
    </main>
  );
}

Run: npm run dev — the UI opens at localhost:3000.
